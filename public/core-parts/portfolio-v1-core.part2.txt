  function weightedAverage(oldAverage, oldLots, newPrice, newLots) {
    var oldAvg = finitePositive(oldAverage, MAX_PRICE);
    var oldLotN = positiveInteger(oldLots, MAX_LOTS);
    var price = finitePositive(newPrice, MAX_PRICE);
    var newLotN = positiveInteger(newLots, MAX_LOTS);
    if (!oldAvg || !oldLotN || !price || !newLotN || oldLotN + newLotN > MAX_LOTS) return null;
    return ((oldAvg * oldLotN) + (price * newLotN)) / (oldLotN + newLotN);
  }

  function calculatePartialSale(position, lotsSold, salePrice) {
    if (!isValidPosition(position)) return { ok: false, error: 'Data posisi tidak valid.' };
    var sold = positiveInteger(lotsSold, position.lots);
    if (!sold) return { ok: false, error: 'Lot jual harus bilangan bulat dan tidak melebihi posisi.' };
    var price = finitePositive(salePrice, MAX_PRICE);
    if (!price) return { ok: false, error: 'Harga jual harus angka positif.' };
    var realized = (price - position.averageBuy) * sold * LOT_SIZE;
    var remainingLots = position.lots - sold;
    return {
      ok: true,
      lotsSold: sold,
      salePrice: price,
      realizedPL: realized,
      remainingLots: remainingLots,
      remainingInvested: position.averageBuy * remainingLots * LOT_SIZE,
      closesPosition: remainingLots === 0,
      remainingAverage: position.averageBuy
    };
  }
