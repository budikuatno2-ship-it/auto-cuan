    function addCalculated() {
      var r = runtime.calculatorResult;
      if (!r || !r.ok) return;
      var existingIndex = runtime.state.positions.findIndex(function (p) { return p.ticker === r.ticker; });
      if (existingIndex >= 0) {
        var existing = runtime.state.positions[existingIndex];
        var avg = weightedAverage(existing.averageBuy, existing.lots, r.buyPrice, r.suggestedLots);
        openModal('Tambah posisi ' + r.ticker,
          '<p class="text-sm text-gray-300">Posisi sudah ada. Tambahan ini akan mengubah rata-rata beli.</p>' +
          summaryCard('Lot Lama', existing.lots + ' lot') + summaryCard('Lot Tambahan', r.suggestedLots + ' lot') + summaryCard('Avg Lama', money(existing.averageBuy)) + summaryCard('Avg Baru', money(avg)),
          function () {
            existing.averageBuy = avg;
            existing.lots += r.suggestedLots;
            existing.investedCapital = avg * existing.lots * LOT_SIZE;
            existing.riskBudget = Math.max(existing.riskBudget || 0, r.budget);
            existing.updatedAt = Date.now();
            saveAndRender('Posisi berhasil ditambahkan.');
          });
        return;
      }
      runtime.state.positions.push(sanitizePosition({
        ticker: r.ticker, lots: r.suggestedLots, averageBuy: r.buyPrice, stop: r.stop,
        target: r.target, riskProfile: r.profile, riskBudget: r.budget,
        createdAt: Date.now(), updatedAt: Date.now(), realizedPL: 0
      }));
      saveAndRender('Posisi berhasil ditambahkan.');
    }
