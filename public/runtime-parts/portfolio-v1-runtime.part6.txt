    function onPositionAction(event) {
      var button = event.target.closest('button[data-action]');
      if (!button) return;
      var card = button.closest('[data-index]');
      var index = Number(card && card.getAttribute('data-index'));
      var position = runtime.state.positions[index];
      if (!position) return;
      var action = button.getAttribute('data-action');
      if (action === 'refresh') refreshTicker(index);
      else if (action === 'add') openAddDialog(index);
      else if (action === 'average') openAverageDialog(index);
      else if (action === 'sell') openSellDialog(index, false);
      else if (action === 'close') openSellDialog(index, true);
      else if (action === 'edit') openEditDialog(index);
      else if (action === 'delete') openDeleteDialog(index);
    }
