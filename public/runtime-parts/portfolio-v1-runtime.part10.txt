  function money(value) {
    if (value == null || !Number.isFinite(Number(value))) return '—';
    var amount = Math.round(Number(value));
    return (amount < 0 ? '-Rp ' : 'Rp ') + Math.abs(amount).toLocaleString('id-ID');
  }

  function pct(value) {
    return value == null || !Number.isFinite(Number(value)) ? '—' : Number(value).toFixed(2) + '%';
  }

  function toast(message, type) {
    if (typeof win.showToast === 'function') win.showToast(message, type || 'success');
    else win.alert(message);
  }

  function positions() {

  function journalKey() {
    var key = typeof win.getPortfolioKey === 'function' ? win.getPortfolioKey() : null;
    return key ? key + '_journal_v1' : null;

  function journalState() {
    var key = journalKey();
    if (!key) return { version: 1, entries: [] };
    var raw = win.localStorage.getItem(key);
    var journalApi = win.AutoCuanPortfolioJournalV1;
    if (journalApi && typeof journalApi.parseState === 'function') return journalApi.parseState(raw);
    try {
      var parsed = JSON.parse(raw || '{}');
      return { version: 1, entries: Array.isArray(parsed.entries) ? parsed.entries : [] };
    } catch (error) {
      return { version: 1, entries: [] };
    }
  }

  function entryForTicker(entries, ticker) {
    var candidates = entries.filter(function (entry) { return entry && entry.ticker === ticker; });
    if (!candidates.length) return null;
    var priority = { active: 0, planned: 1, finished: 2, cancelled: 3 };
    candidates.sort(function (a, b) {
      var aRank = priority[a.status] == null ? 9 : priority[a.status];
      var bRank = priority[b.status] == null ? 9 : priority[b.status];
      if (aRank !== bRank) return aRank - bRank;
      return Number(b.updatedAt || 0) - Number(a.updatedAt || 0);
    });
    return candidates[0];
  }

  function buildReport() {
    var current = positions();
    var entries = journalState().entries;
    var summary = api.portfolioSummary(current);
    var largest = null;
    var covered = 0;
    var warnings = [];
    var activeTickers = new Set();

    current.forEach(function (position) {
      activeTickers.add(position.ticker);
      var metrics = api.positionMetrics(position);
      var concentration = summary.invested > 0 ? metrics.invested / summary.invested * 100 : 0;
      if (!largest || concentration > largest.concentration) largest = { ticker: position.ticker, concentration: concentration };

      var journal = entryForTicker(entries, position.ticker);
      if (journal && journal.status !== 'cancelled') covered += 1;
      else warnings.push({ tone: 'amber', text: position.ticker + ': posisi aktif belum memiliki jurnal yang dapat dipakai.' });

      var rate = api.RISK_PROFILES[position.riskProfile] || 0;
      var allowedRisk = Number(position.riskBudget || 0) * rate;
      if (allowedRisk > 0 && metrics.estimatedLossAtStop > allowedRisk * 1.05) {
        warnings.push({ tone: 'amber', text: position.ticker + ': risiko ke stop ' + money(metrics.estimatedLossAtStop) + ' melebihi batas profil awal ' + money(allowedRisk) + '.' });
      }
      if (metrics.priced && position.lastPrice <= position.stop) {
        warnings.push({ tone: 'red', text: position.ticker + ': harga terakhir sudah sama dengan atau di bawah stop/invalidation.' });
      } else if (metrics.priced && metrics.distanceCurrentToStopPct != null && metrics.distanceCurrentToStopPct <= 3) {
        warnings.push({ tone: 'amber', text: position.ticker + ': jarak ke stop tinggal ' + pct(metrics.distanceCurrentToStopPct) + '.' });
      }
      if (position.priceStale) warnings.push({ tone: 'gray', text: position.ticker + ': harga bertanda T-1/stale.' });
      if (!metrics.priced) warnings.push({ tone: 'gray', text: position.ticker + ': harga belum tersedia sehingga total nilai belum lengkap.' });
    });

    entries.forEach(function (entry) {
      if (entry.status === 'active' && !activeTickers.has(entry.ticker)) {
        warnings.push({ tone: 'gray', text: entry.ticker + ': jurnal berstatus Aktif tetapi ticker belum ada di Posisi Aktif.' });
      }
    });

    if (largest && largest.concentration >= 50) {
      warnings.unshift({ tone: 'amber', text: largest.ticker + ' mencakup ' + pct(largest.concentration) + ' dari modal aktif.' });
    }
    var totalRiskPct = summary.invested > 0 ? summary.estimatedRisk / summary.invested * 100 : null;
    if (totalRiskPct != null && totalRiskPct > 5) {
      warnings.unshift({ tone: 'amber', text: 'Estimasi risiko seluruh posisi ke stop mencapai ' + pct(totalRiskPct) + ' dari modal aktif.' });
    }

    return {
      positions: current,
      summary: summary,
      largest: largest,
      covered: covered,
      coveragePct: current.length ? covered / current.length * 100 : null,
      totalRiskPct: totalRiskPct,
      warnings: warnings.slice(0, 8)
    };
  }
