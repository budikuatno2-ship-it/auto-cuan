        },
        function () {
          doc.getElementById('pfAvgSimulate').addEventListener('click', function () {
            var sim = simulateAverageDown(p, { price: doc.getElementById('pfAvgPrice').value, lots: doc.getElementById('pfAvgLots').value, budget: doc.getElementById('pfAvgBudget').value });

            var result = doc.getElementById('pfAvgResult');
            if (!sim.ok) { result.innerHTML = '<p class="text-red-300">' + escapeHtml(sim.error) + '</p>'; return; }

'<p class="text-amber-300">Risiko melebihi profil ' + escapeHtml(p.riskProfile) + '.</p>' : '') + (sim.invalidated ? '<p class="text-red-300 font-semibold">Average down tidak disarankan karena harga telah melewati batas invalidasi.</p>' : '<p class="text-gray-500">Simulasi matematika, bukan rekomendasi pembelian.</p>');
            doc.getElementById('pfModalConfirm').disabled = !sim.canApply;
          });
        });
    }
